function EditableText({ value, onChange, className }) {
  const [editing, setEditing] = useState(false);
  const [val, setVal] = useState(value);
  const ref = useRef();

  const start = (e) => {
    e.stopPropagation();
    setEditing(true);
    setTimeout(() => ref.current?.select(), 10);
  };

  const finish = () => {
    setEditing(false);
    onChange(val);
  };

  if (editing) {
    return (
      <input
        ref={ref}
        value={val}
        onChange={(e) => setVal(e.target.value)}
        onBlur={finish}
        onKeyDown={(e) => e.key === "Enter" && finish()}
        className={`bg-transparent border-b border-white/60 outline-none text-center w-full ${className}`}
        style={{ minWidth: 60 }}
      />
    );
  }
  return (
    <span
      className={`cursor-pointer hover:underline hover:decoration-dotted ${className}`}
      onDoubleClick={start}
      title="Klik dua kali untuk edit"
    >
      {value}
    </span>
  );
}

function OrgChart({ data, onUpdate }) {
  const topNodes = data.children.filter((c) => c.tier === "top");
  const ketua = data.children.find((c) => c.id === "ketua");

  const updateRoot = useCallback((updated) => onUpdate(updated), [onUpdate]);
  const updateChild = useCallback((updated) => {
    const newChildren = data.children.map((c) => (c.id === updated.id ? updated : c));
    onUpdate({ ...data, children: newChildren });
  }, [data, onUpdate]);
  const updateKetua = useCallback((updated) => {
    const newChildren = data.children.map((c) => (c.id === updated.id ? updated : c));
    onUpdate({ ...data, children: newChildren });
  }, [data, onUpdate]);
  const updateKetauChild = useCallback((updated) => {
    const newKetauChildren = ketua.children.map((c) => (c.id === updated.id ? updated : c));
    const newKetua = { ...ketua, children: newKetauChildren };
    const newChildren = data.children.map((c) => (c.id === newKetua.id ? newKetua : c));
    onUpdate({ ...data, children: newChildren });
  }, [data, ketua, onUpdate]);

  const wk1 = ketua?.children.find((c) => c.id === "wk1");
  const wk2 = ketua?.children.find((c) => c.id === "wk2");
  const wk3 = ketua?.children.find((c) => c.id === "wk3");
  const sekjen = ketua?.children.find((c) => c.id === "sekjen");
  const bendahara = ketua?.children.find((c) => c.id === "bendahara");
  const unitPendukung = ketua?.children.find((c) => c.id === "unit-pendukung");

  return (
    <div className="flex flex-col items-center gap-0 w-full">
      {/* Root */}
      <div className="flex flex-col items-center">
        <Node node={data} onUpdate={updateRoot} />
        <div className="w-px bg-amber-400/60 h-6" />
      </div>

      {/* Top tier: Pelindung, Dewan Pengarah, Dewan Penasihat */}
      <div className="flex flex-row gap-4 items-center">
        {topNodes.map((n) => (
          <Node key={n.id} node={n} onUpdate={updateChild} />
        ))}
      </div>
      <div className="w-px bg-slate-500 h-6" />

      {/* Ketua Umum */}
      <Node node={ketua} onUpdate={updateKetua} />
      <div className="w-px bg-cyan-500/60 h-6" />

      {/* Main row */}
      <div className="flex flex-row items-start gap-5 flex-wrap justify-center">
        {/* WK1 */}
        <WakilBlock node={wk1} onUpdate={updateKetauChild} accentColor="border-indigo-400" />
        {/* Staff */}
        <div className="flex flex-col gap-3 items-center justify-start mt-0">
          {sekjen && <Node node={sekjen} onUpdate={updateKetauChild} />}
          {bendahara && <Node node={bendahara} onUpdate={updateKetauChild} />}
        </div>
        {/* WK2 */}
        <WakilBlock node={wk2} onUpdate={updateKetauChild} accentColor="border-indigo-400" />
        {/* WK3 */}
        <WakilBlock node={wk3} onUpdate={updateKetauChild} accentColor="border-indigo-400" />
        {/* Unit Pendukung */}
        <WakilBlock node={unitPendukung} onUpdate={updateKetauChild} accentColor="border-emerald-400" />
      </div>
    </div>
  );
}

function WakilBlock({ node, onUpdate, accentColor }) {

  return (
    <div className="flex flex-col items-center">

      <div className="w-px bg-slate-500 h-4" />

        {node.children?.map((child) => (
          <Node key={child.id} node={child} onUpdate={updateChild} />
        ))}
      </div>
    </div>
  );
}

function EditableText({ value, onChange, className, style }) {
  const [editing, setEditing] = useState(false);
  const [val, setVal] = useState(value);
  const ref = useRef();
  const start = (e) => { e.stopPropagation(); setEditing(true); setTimeout(() => ref.current?.select(), 10); };
  const finish = () => { setEditing(false); onChange(val); };
  if (editing) return (
    <input ref={ref} value={val} onChange={e => setVal(e.target.value)}
      onBlur={finish} onKeyDown={e => e.key === "Enter" && finish()}
      className={`bg-transparent border-b border-white/60 outline-none text-center w-full ${className}`} style={style} />
  );
  return <span className={`cursor-pointer hover:underline hover:decoration-dotted ${className}`} style={style} onDoubleClick={start} title="Klik dua kali untuk edit">{value}</span>;
}

function NodeBox({ node, onUpdate }) {
  const s = node.color || tierStyles[node.tier] || tierStyles.unit;
  const upd = f => v => onUpdate({ ...node, [f]: v });
  return (
    <div className={`rounded-xl border ${s.border} bg-gradient-to-br ${s.bg} px-3 py-2 min-w-[130px] max-w-[185px] text-center relative group`}
      style={{ boxShadow: s.glow }}>
      <EditableText value={node.label} onChange={upd("label")} className={`font-bold text-xs leading-tight block ${s.text}`} />
      {node.sublabel && <EditableText value={node.sublabel} onChange={upd("sublabel")} className={`text-[10px] opacity-75 mt-0.5 block ${s.text}`} />}
      <span className="absolute -top-2 -right-2 hidden group-hover:flex w-4 h-4 rounded-full bg-white/20 text-white text-[8px] items-center justify-center">✎</span>
    </div>
  );
}

function BidangColumn({ node, onUpdate }) {
  const updateChild = (updated) => {
    const newChildren = node.children.map(c => c.id === updated.id ? updated : c);
    onUpdate({ ...node, children: newChildren });
  };
  const s = node.color || tierStyles[node.tier];
  return (
    <div className="flex flex-col items-center">
      <NodeBox node={node} onUpdate={onUpdate} />
      <div className="w-px h-4" style={{ background: "#64748b" }} />
      <div className={`flex flex-col gap-2 border-l-2 border-r-2 border-t-2 ${s.border} rounded-t-lg px-2 pt-2 pb-0`}>
        {node.children.map(child => (
          <NodeBox key={child.id} node={child} onUpdate={updateChild} />
        ))}
      </div>
    </div>
  );
}

function OrgChart({ data, onUpdate }) {
  const ketua = data.children[0];
  const updateKetua = (updated) => onUpdate({ ...data, children: [updated] });
  const updateBidang = (updated) => {
    const newChildren = ketua.children.map(c => c.id === updated.id ? updated : c);
    updateKetua({ ...ketua, children: newChildren });
  };

  return (
    <div className="flex flex-col items-center">
      <NodeBox node={data} onUpdate={upd => onUpdate({ ...data, ...upd })} />
      <div className="w-px h-6" style={{ background: "#f59e0b88" }} />
      <NodeBox node={ketua} onUpdate={updateKetua} />
      <div className="w-px h-6" style={{ background: "#22d3ee88" }} />
      <div className="flex flex-row items-start gap-4 flex-wrap justify-center">
        {ketua.children.map(bidang => (
          <BidangColumn key={bidang.id} node={bidang} onUpdate={updateBidang} />
        ))}
      </div>
    </div>
  );
}
