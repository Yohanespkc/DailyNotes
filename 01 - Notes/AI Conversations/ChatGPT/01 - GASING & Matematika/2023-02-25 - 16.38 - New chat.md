```javascript
function multiplyMatrix(matrix1, matrix2) {
  let result = [];
  for (let i = 0; i < 3; i++) {
    result[i] = [];
    for (let j = 0; j < 3; j++) {
      let sum = 0;
      for (let k = 0; k < 3; k++) {

      }

    }
  }

}

```
