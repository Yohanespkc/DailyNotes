```python
import random

def generate_question():
    num1 = random.randint(0, 9)
    num2 = random.randint(0, 9)
    correct_answer = num1 + num2
    question = f"Berapakah {num1} + {num2}?"
    return question, str(correct_answer)

def check_answer(user_answer, correct_answer):
    if user_answer == correct_answer:
        print("Jawaban Anda benar!")
        return True
    else:
        print("Jawaban Anda salah.")
        return False

def play_game():
    print("Selamat datang di game penjumlahan 1 digit!")
    print("Jawablah pertanyaan dengan benar untuk mencetak skor tinggi.")
    print("Ketik 'exit' untuk keluar dari permainan.\n")

    score = 0
    while True:
        question, correct_answer = generate_question()
        print(question)
        user_answer = input("Jawaban Anda: ")

        if user_answer.lower() == "exit":
            break

        if check_answer(user_answer, correct_answer):
            score += 1
            print("Skor Anda:", score)
        else:
            print("Skor Anda:", score)

    print("Terima kasih telah bermain! Skor akhir Anda adalah:", score)

play_game()
```

Game ini akan menghasilkan pertanyaan penjumlahan dua angka acak antara 0 hingga 9. Anda harus menjawab dengan memasukkan jawaban numerik. Setiap jawaban yang benar akan meningkatkan skor Anda. Ketik "exit" untuk keluar dari permainan. Saat Anda keluar, permainan akan mencetak skor akhir Anda.
